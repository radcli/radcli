60/*- Select the server list and rc_type for a request, based on transport
61 * and request type. Internal helper shared by rc_aaa_ctx() and
62 * rc_acct_async().
63 *
64 * @param rh a handle to parsed configuration.
65 * @param aaaserver receives the selected SERVER list from configuration.
66 * @param type receives AUTH or ACCT, matching the selected server list.
67 * @param request_type one of the standard RADIUS codes (e.g., PW_ACCESS_REQUEST).
68 * @return OK_RC (0) on success, ERROR_RC if no matching servers are configured.
69 -*/
70static int rc_select_aaa_server(rc_handle *rh, SERVER **aaaserver,
71 rc_type *type, rc_standard_codes request_type)
72{
73 if (RC_USES_SESSION(rh) || request_type != PW_ACCOUNTING_REQUEST) {
74 *aaaserver = rc_conf_srv(rh, "authserver");
75 *type = AUTH;
76 } else {
77 *aaaserver = rc_conf_srv(rh, "acctserver");
78 *type = ACCT;
79 }
80
81 if (*aaaserver == NULL)
82 return ERROR_RC;
83
84 return OK_RC;
85}

87/*- Fill in NAS-Port and Acct-Delay-Time on a request being built.
88 * Internal helper shared by rc_aaa_ctx_server() and
89 * rc_aaa_ctx_server_async().
90 *
91 * @param rh a handle to parsed configuration.
92 * @param data the request being built; send_pairs is extended in place.
93 * @param nas_port the physical NAS port number to include (may be zero).
94 * @param add_nas_port if non-zero, PW_NAS_PORT is added to the sent pairs.
95 * @param request_type one of the standard RADIUS codes (e.g., PW_ACCESS_REQUEST).
96 * @param adt_vp receives the Acct-Delay-Time pair when request_type is
97 * PW_ACCOUNTING_REQUEST, so the caller can update it on each retransmission.
98 * @param start_time receives the time the delay is measured from.
99 * @return OK_RC (0) on success, ERROR_RC on failure.
100 -*/
101static int rc_fill_acct_pairs(rc_handle const *rh, SEND_DATA *data,
102 uint32_t nas_port, int add_nas_port,
103 rc_standard_codes request_type,
104 VALUE_PAIR **adt_vp, double *start_time)
105{
106 time_t dtime;
107 double now;
108
109 if (add_nas_port != 0
110 && rc_avpair_get(data->send_pairs, PW_NAS_PORT, 0) == NULL) {
111 /*
112 * Fill in NAS-Port
113 */
114 if (rc_avpair_add(rh, &(data->send_pairs), PW_NAS_PORT,
115 &nas_port, 0, 0) == NULL)
116 return ERROR_RC;
117 }
118
119 if (request_type == PW_ACCOUNTING_REQUEST) {
120 /*
121 * Fill in Acct-Delay-Time
122 */
123 dtime = 0;
124 now = rc_getmtime();
125 *adt_vp = rc_avpair_get(data->send_pairs, PW_ACCT_DELAY_TIME, 0);
126 if (*adt_vp == NULL) {
127 *adt_vp = rc_avpair_add(rh, &(data->send_pairs),
128 PW_ACCT_DELAY_TIME, &dtime, 0,
129 0);
130 if (*adt_vp == NULL)
131 return ERROR_RC;
132 *start_time = now;
133 } else {
134 *start_time = now - (*adt_vp)->lvalue;
135 }
136 }
137
138 return OK_RC;
139}

204int rc_aaa_ctx_server(rc_handle * rh, RC_AAA_CTX ** ctx, SERVER * aaaserver,
205 rc_type type,
206 uint32_t nas_port,
207 VALUE_PAIR * send, VALUE_PAIR ** received,
208 char *msg, int add_nas_port,
209 rc_standard_codes request_type)
210{
211 SEND_DATA data;
212 VALUE_PAIR *adt_vp = NULL;
213 int result;
214 int timeout = rc_conf_int_id(rh, OPT_RADIUS_TIMEOUT);
215 int retries = rc_conf_int_id(rh, OPT_RADIUS_RETRIES);
216 double start_time = 0;
217 time_t dtime;
218 int servernum;
219
220 data.send_pairs = send;
221 data.receive_pairs = NULL;
222
223 if (rc_fill_acct_pairs(rh, &data, nas_port, add_nas_port, request_type,
224 &adt_vp, &start_time) != OK_RC)
225 return ERROR_RC;
226
227 if (data.receive_pairs != NULL) {
228 rc_avpair_free(data.receive_pairs);
229 data.receive_pairs = NULL;
230 }
231
232 servernum = 0;
233 do {
234 rc_buildreq(rh, &data, request_type, aaaserver->name[servernum],
235 aaaserver->port[servernum],
236 aaaserver->secret[servernum], timeout, retries);
237
238 if (request_type == PW_ACCOUNTING_REQUEST) {
239 dtime = rc_getmtime() - start_time;
240 rc_avpair_assign(adt_vp, &dtime, 0);
241 }
242
243 result = rc_send_server_ctx(rh, ctx, &data, msg, type, 0);
244
245 if ((result == OK_RC) || (result == CHALLENGE_RC) || (result == REJECT_RC)) {
246 if (request_type != PW_ACCOUNTING_REQUEST) {
247 *received = data.receive_pairs;
248 } else {
249 rc_avpair_free(data.receive_pairs);
250 }
251
252 DEBUG(rh, LOG_INFO,
253 "rc_send_server_ctx returned success for server %u", servernum);
254 return result;
255 }
256
257 rc_avpair_free(data.receive_pairs);
258 data.receive_pairs = NULL;
259
260 DEBUG(rh, LOG_INFO, "rc_send_server_ctx returned error (%d) for server %u: (remaining: %d)",
261 result, servernum, aaaserver->max-servernum);
262 servernum++;
263 } while (servernum < aaaserver->max && ((result == TIMEOUT_RC) || (result == NETUNREACH_RC)));
264
265 return result;
266}

360/*- Send an accounting request to every server in aaaserver without
361 * waiting for a reply. Internal helper behind rc_acct_async().
362 *
363 * @param rh a handle to parsed configuration.
364 * @param aaaserver a non-NULL SERVER describing the target server(s).
365 * @param type AUTH or ACCT, selects the destination port.
366 * @param nas_port the physical NAS port number to include (may be zero).
367 * @param send VALUE_PAIR list of attributes to send.
368 * @return OK_RC (0) if the packet was handed to the socket layer for at
369 * least one server, ERROR_RC on failure.
370 -*/
371static int rc_aaa_ctx_server_async(rc_handle * rh, SERVER * aaaserver,
372 rc_type type, uint32_t nas_port,
373 VALUE_PAIR * send)
374{
375 SEND_DATA data;
376 VALUE_PAIR *adt_vp = NULL;
377 double start_time = 0;
378 time_t dtime;
379 int timeout = rc_conf_int_id(rh, OPT_RADIUS_TIMEOUT);
380 int retries = rc_conf_int_id(rh, OPT_RADIUS_RETRIES);
381 int servernum;
382 int result;
383 int sent = 0;
384
385 data.send_pairs = send;
386 data.receive_pairs = NULL;
387
388 if (rc_fill_acct_pairs(rh, &data, nas_port, 1, PW_ACCOUNTING_REQUEST,
389 &adt_vp, &start_time) != OK_RC)
390 return ERROR_RC;
391
392 for (servernum = 0; servernum < aaaserver->max; servernum++) {
393 dtime = rc_getmtime() - start_time;
394 rc_avpair_assign(adt_vp, &dtime, 0);
395
396 /* timeout/retries are not consulted by rc_send_server_ctx()
397 * when no_wait is set below; passed through unchanged only
398 * so SEND_DATA/DEBUG output reflect the real configuration. */
399 rc_buildreq(rh, &data, PW_ACCOUNTING_REQUEST,
400 aaaserver->name[servernum],
401 aaaserver->port[servernum],
402 aaaserver->secret[servernum], timeout, retries);
403
404 result = rc_send_server_ctx(rh, NULL, &data, NULL, type, 1);
405
406 if (data.receive_pairs != NULL) {
407 rc_avpair_free(data.receive_pairs);
408 data.receive_pairs = NULL;
409 }
410
411 if (result == OK_RC) {
412 sent++;
413 } else {
414 DEBUG(rh, LOG_INFO,
415 "rc_send_server_ctx returned error (%d) for server %u",
416 result, servernum);
417 }
418 }
419
420 return sent > 0 ? OK_RC : ERROR_RC;
421}

462int rc_check(rc_handle * rh, char *host, char *secret, unsigned short port,
463 char *msg)
464{
465 SEND_DATA data;
466 int result;
467 uint32_t service_type;
468 int timeout = rc_conf_int_id(rh, OPT_RADIUS_TIMEOUT);
469 int retries = rc_conf_int_id(rh, OPT_RADIUS_RETRIES);
470 rc_type type;
471
472 data.send_pairs = data.receive_pairs = NULL;
473
474 if (RC_USES_SESSION(rh))
475 type = AUTH;
476 else
477 type = ACCT;
478
479 /*
480 * Fill in Service-Type
481 */
482
483 service_type = PW_ADMINISTRATIVE;
484 rc_avpair_add(rh, &(data.send_pairs), PW_SERVICE_TYPE, &service_type, 0,
485 0);
486
487 rc_buildreq(rh, &data, PW_STATUS_SERVER, host, port, secret, timeout,
488 retries);
489 result = rc_send_server(rh, &data, msg, type);
490
491 rc_avpair_free(data.receive_pairs);
492
493 return result;
494}
